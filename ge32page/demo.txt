        // 解析带**标记的文本
        function parseMarkedText(text) {
            // 使用正则表达式匹配**包裹的内容
            return text.replace(/\*\*(.*?)\*\*/g, '<span class="highlight">$1</span>');
        }

        // 创建单个卡片元素
        function createCardElement(paragraph) {
            const card = document.createElement('div');
            card.className = 'card';
            
            const cardHeader = document.createElement('div');
            cardHeader.className = 'card-header';
            
            const sectionIndex = document.createElement('div');
            sectionIndex.className = 'section-index';
            sectionIndex.textContent = `Section ${paragraph.index}`;
            
            const levelBadge = document.createElement('div');
            levelBadge.className = 'level-badge';
            levelBadge.textContent = paragraph.level;
            
            cardHeader.appendChild(sectionIndex);
            cardHeader.appendChild(levelBadge);
            card.appendChild(cardHeader);
            
            const contentEn = document.createElement('div');
            contentEn.className = 'content-en';
            contentEn.innerHTML = parseMarkedText(paragraph.content_en);
            card.appendChild(contentEn);
            
            const contentCn = document.createElement('div');
            contentCn.className = 'content-cn';
            contentCn.innerHTML = parseMarkedText(paragraph.content_cn);
            card.appendChild(contentCn);
            
            // 添加术语部分
            if (paragraph.terms && paragraph.terms.length > 0) {
                const termsContainer = document.createElement('div');
                termsContainer.className = 'terms-container';
                
                const termsTitle = document.createElement('div');
                termsTitle.className = 'terms-title';
                termsTitle.innerHTML = '<i class="fas fa-gem"></i> 核心术语';
                termsContainer.appendChild(termsTitle);
                
                const termsGrid = document.createElement('div');
                termsGrid.className = 'terms-grid';
                
                paragraph.terms.forEach(term => {
                    const termCard = document.createElement('div');
                    termCard.className = 'term-card';
                    
                    const termWord = document.createElement('div');
                    termWord.className = 'term-word';
                    termWord.textContent = term.word;
                    termCard.appendChild(termWord);
                    
                    const termDefinition = document.createElement('div');
                    termDefinition.className = 'term-definition';
                    termDefinition.textContent = term.definition;
                    termCard.appendChild(termDefinition);
                    
                    const exampleEn = document.createElement('div');
                    exampleEn.className = 'term-example';
                    exampleEn.textContent = term.sentence_en;
                    termCard.appendChild(exampleEn);
                    
                    const exampleCn = document.createElement('div');
                    exampleCn.className = 'term-example';
                    exampleCn.textContent = term.sentence_cn;
                    termCard.appendChild(exampleCn);
                    
                    termsGrid.appendChild(termCard);
                });
                
                termsContainer.appendChild(termsGrid);
                card.appendChild(termsContainer);
            }
            
            return card;
        }

        // 初始化页面
        function initPage() {
            // 设置页面标题
            document.getElementById('pageTitle').textContent = pageData.title;
            document.getElementById('pageSubtitle').textContent = "远程工作如何重塑全球城市 - 科技风数据可视化展示";
            
            // 获取内容容器
            const contentContainer = document.getElementById('contentContainer');
            
            // 清空容器
            contentContainer.innerHTML = '';
            
            // 创建并添加卡片
            pageData.paragraphs.forEach(paragraph => {
                const card = createCardElement(paragraph);
                contentContainer.appendChild(card);
            });
            
            // 添加动画效果
            const cards = document.querySelectorAll('.card');
            cards.forEach((card, index) => {
                card.style.animationDelay = `${index * 0.1}s`;
            });
            
            // 隐藏加载层
            setTimeout(() => {
                document.getElementById('loadingOverlay').style.opacity = '0';
                setTimeout(() => {
                    document.getElementById('loadingOverlay').style.display = 'none';
                }, 500);
            }, 1000);
        }

        // 页面加载完成后初始化
        document.addEventListener('DOMContentLoaded', initPage);
